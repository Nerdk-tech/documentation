from flask import Flask, request, jsonifyfrom flask import Flask, request, jsonify
import subprocess
import os

app = Flask(__name__)

# Directory for bots
BOT_DIR = os.path.expanduser("~/bots")

# Ensure bot directory exists
if not os.path.exists(BOT_DIR):
    os.makedirs(BOT_DIR)

@app.route("/")
def home():
    return "Pterodactyl-like Panel Running!"

@app.route("/deploy", methods=["POST"])
def deploy_bot():
    data = request.json
    bot_name = data.get("name", "my_bot")
    script = data.get("script", "print('Hello from the bot!')")

    bot_path = os.path.join(BOT_DIR, f"{bot_name}.py")
    
    # Save script
    with open(bot_path, "w") as f:
        f.write(script)

    return jsonify({"message": f"Bot {bot_name} deployed!", "path": bot_path})

@app.route("/start", methods=["POST"])
def start_bot():
    data = request.json
    bot_name = data.get("name")

    bot_path = os.path.join(BOT_DIR, f"{bot_name}.py")

    if not os.path.exists(bot_path):
        return jsonify({"error": "Bot not found"}), 404

    # Run bot in the background
    subprocess.Popen(["python", bot_path], stdout=subprocess.PIPE, stderr=subprocess.PIPE)

    return jsonify({"message": f"Bot {bot_name} started!"})

@app.route("/list", methods=["GET"])
def list_bots():
    bots = [f for f in os.listdir(BOT_DIR) if f.endswith(".py")]
    return jsonify({"bots": bots})

@app.route("/stop", methods=["POST"])
def stop_bot():
    return jsonify({"message": "Stopping bots is not implemented yet!"})

if __name__ == "__main__":
